[\[7\]](https://team-workspace-jf9n.dovetail.com/data/1eR1c9MHYjKPjF74thvn7m?utm_medium=citation#:l:q=bmFudGkgZW5naW5lZXIgbmdlcmphaW4gYmFsaWsga2Ugc2F5YSBiYWxpayBrZSBzYXlhLiBLYWxhdSB1ZGFoIHNlbGVzYWkgc2VtdWEgc2F5YSBtaW50YSB0YW5kYSB0YW5nYW4gZGFyaSBDUC4gSGFiaXMgaXR1IGhhYmlzIGl0dSBsYW5nc3VuZyBkaXRhcnVoIGRpIHJlY29yZCBkaXNpbXBhbiBzYW1hIGRpYSByZWNvcmQu&s=0&e=183)("nanti engineer ngerjain balik ke saya balik ke saya.
